cmake_minimum_required(VERSION 3.21)
project(VeyraDesktop VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebEngineWidgets)
qt_add_executable(veyra src/main.cpp)
target_link_libraries(veyra PRIVATE Qt6::Widgets Qt6::WebEngineWidgets)
